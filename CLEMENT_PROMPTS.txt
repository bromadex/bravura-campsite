export default function PageName() {
  const { currentSiteId } = useSite()
  const { can } = usePermissions()
  const [data, setData] = useState(null)
  const [loading, setLoading] = useState(true)

  if (!can('fuel.view')) return <AccessDenied />

  useEffect(() => {
    if (!currentSiteId) return
    async function load() {
      const { supabase } = await import('../../supabaseClient')
      const { data, error } = await supabase
        .from('table_name')
        .select('*')
        .eq('site_id', currentSiteId)
        .eq('is_archived', false)
        .order('created_at', { ascending: false })
      if (error) { console.error(error); return }
      setData(data)
      setLoading(false)
    }
    load()
  }, [currentSiteId])

  if (loading) return <LoadingSpinner />
  return ( ... )
}

── FUEL TRANSACTION RULES ──────────────────────────────────────────────────────

Fuel transactions are IMMUTABLE after creation.
Never update or delete a fuel_transaction row.
If a correction is needed, create a new transaction with type='adjustment'
referencing the original transaction.

Tank levels are updated by trigger, not by application code directly:
  - Each INSERT into fuel_transactions triggers an update to fuel_tanks.current_level_litres
  - issuance: current_level -= litres
  - delivery: current_level += litres
  - adjustment: current_level += adjustment_litres (can be negative)

fuel_types(
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  name TEXT NOT NULL,           -- 'Diesel', 'Petrol', 'Paraffin', 'Aviation'

  colour TEXT,                  -- display colour for charts/badges

  created_at TIMESTAMPTZ DEFAULT now()

fuel_tanks(
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  site_id UUID NOT NULL REFERENCES sites(id),
  name TEXT NOT NULL,
  code TEXT NOT NULL,
  tank_type TEXT NOT NULL,       -- 'underground', 'above_ground', 'bowser', 'mobile'
  fuel_type_id UUID NOT NULL REFERENCES fuel_types(id),
  capacity_litres NUMERIC(12,3) NOT NULL,
  current_level_litres NUMERIC(12,3) NOT NULL DEFAULT 0,
  min_threshold_litres NUMERIC(12,3) DEFAULT 0,
  min_threshold_percent NUMERIC(5,2) DEFAULT 20,
  location_description TEXT,
  gps_lat NUMERIC(10,7),
  gps_lng NUMERIC(10,7),
  atg_device_id TEXT,            -- future IoT integration hook
  last_dip_date DATE,
  last_dip_reading NUMERIC(12,3),
  status TEXT DEFAULT 'active',  -- 'active', 'maintenance', 'decommissioned'
  is_archived BOOLEAN DEFAULT false,
  archived_at TIMESTAMPTZ,
  created_at TIMESTAMPTZ DEFAULT now(),
  updated_at TIMESTAMPTZ DEFAULT now(),
  created_by UUID REFERENCES profiles(id),
  UNIQUE(site_id, code),
  CONSTRAINT positive_level CHECK (current_level_litres >= 0),
  CONSTRAINT level_within_capacity CHECK (current_level_litres <= capacity_litres)
)

fuel_pumps(

  status TEXT DEFAULT 'active',   -- 'active', 'maintenance', 'decommissioned'

  UNIQUE(site_id, code)
)

fuel_vehicles(

  archived_at TIMESTAMPTZ,
  created_at TIMESTAMPTZ DEFAULT now(),

fuel_equipment(
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,

  expected_consumption_lph NUMERIC(6,3),

fuel_operators(
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  site_id UUID NOT NULL REFERENCES sites(id),
  employee_id UUID NOT NULL REFERENCES employees(id),

  activated_at TIMESTAMPTZ DEFAULT now(),

  created_at TIMESTAMPTZ DEFAULT now(),

fuel_transactions(
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  site_id UUID NOT NULL REFERENCES sites(id),
  transaction_number TEXT NOT NULL,
  tank_id UUID NOT NULL REFERENCES fuel_tanks(id),
  pump_id UUID REFERENCES fuel_pumps(id),
  transaction_type TEXT NOT NULL,     -- 'issuance', 'delivery', 'adjustment', 'dip_correction'
  vehicle_id UUID REFERENCES fuel_vehicles(id),
  equipment_id UUID REFERENCES fuel_equipment(id),
  operator_id UUID REFERENCES fuel_operators(id),
  litres NUMERIC(12,3) NOT NULL,
  meter_start NUMERIC(12,3),
  meter_end NUMERIC(12,3),
  tank_level_before NUMERIC(12,3),
  tank_level_after NUMERIC(12,3),
  unit_price NUMERIC(10,4),
  total_cost NUMERIC(15,2),
  fuel_request_id UUID,               -- FK to fuel_requests (Phase 2)
  original_transaction_id UUID,       -- FK to self, for adjustments
  approved_by UUID REFERENCES profiles(id),
  approved_at TIMESTAMPTZ,
  docket_number TEXT,
  notes TEXT,
  created_at TIMESTAMPTZ DEFAULT now(),
  created_by UUID REFERENCES profiles(id),
  UNIQUE(site_id, transaction_number)
)
